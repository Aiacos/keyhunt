```cpp
TEST(intgroup_batch_vs_single_inversion) {
    // Create N random-ish values
    const int N = 8;
    Int values[N];
    Int batch_results[N];
    Int single_results[N];

    // Initialize with deterministic non-zero values
    for (int i = 0; i < N; i++) {
        values[i].SetInt32(100 + i * 37);  // Deterministic, non-zero
        batch_results[i] = values[i];
        single_results[i] = values[i];
    }

    // Batch inversion via IntGroup
    IntGroup group(N);
    group.Set(batch_results);
    group.ModInv();

    // Single-element inversion for each
    for (int i = 0; i < N; i++) {
        single_results[i].ModInv();
    }

    // Compare: batch result must equal single result for each element
    for (int i = 0; i < N; i++) {
        ASSERT_TRUE(batch_results[i].IsEqual(&single_results[i]));
    }
}
```
